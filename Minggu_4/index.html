def ukuran_mentah(lebar, tinggi, bit):        # persis slide 12
    return lebar * tinggi * bit // 8

def kedalaman_bit(citra, n_bit):               # persis slide 10
    tingkat = 2 ** n_bit
    langkah = 256 // tingkat
    return (citra // langkah) * langkah
