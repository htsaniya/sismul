def rgb_ke_ycbcr(r, g, b):     # rentang studio BT.601, persis slide 24
    Y  =  0.257*r + 0.504*g + 0.098*b + 16
    Cb = -0.148*r - 0.291*g + 0.439*b + 128
    Cr =  0.439*r - 0.368*g - 0.071*b + 128
    return Y, Cb, Cr

def rgb_ke_cmyk(r, g, b):       # r,g,b dinormalisasi 0-1, persis slide 22
    c, m, y = 1-r, 1-g, 1-b
    k = min(c, m, y)
    return c-k, m-k, y-k, k
